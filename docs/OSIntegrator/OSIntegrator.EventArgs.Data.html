АргументыСобытия.Данные (EventArgs.Data) Свойство

Использование

Только чтение.

Значение

Тип: Строка.

Возвращает данные из потоков процесса.

Примечание

Пример

Сообщить("" + ИОС.АргументыСобытия.Данные);
Полный пример кода 



Перем ИОС, Процесс1, ИнформацияЗапускаПроцесса1;

Процедура ОбработкаТаймера1() Экспорт
	Сообщить("Дескриптор = " + Процесс1.Дескриптор);
	
	Сообщить("Принудительная остановка процесса...");
	Если Не Процесс1.Закрыт Тогда
		Процесс1.ОтменитьЧтениеВывода();
		Процесс1.Уничтожить();
		Приостановить(1000);
		ИОС.Продолжать = Ложь;
	КонецЕсли;
КонецПроцедуры

Процедура Процесс1_ПриПолученииДанных() Экспорт
	Сообщить("" + ИОС.АргументыСобытия.Данные);
КонецПроцедуры

ПодключитьВнешнююКомпоненту(ОбъединитьПути(Новый Файл(ТекущийСценарий().Источник).Путь, "OneScriptIntegrator.dll"));
ИОС = Новый ИнтеграторОдноСкрипта();

Стр = "
|Процедура ОбработкаТаймера() Экспорт
|	Сообщить(""Таймер "" + ТекущаяУниверсальнаяДатаВМиллисекундах());
|КонецПроцедуры
|
|ПодключитьВнешнююКомпоненту(ОбъединитьПути(Новый Файл(ТекущийСценарий().Источник).Путь, ""OneScriptIntegrator.dll""));
|ИОС = Новый ИнтеграторОдноСкрипта();
|
|Таймер = ИОС.Таймер();
|Таймер.Начать(ИОС.Действие(ЭтотОбъект, ""ОбработкаТаймера""), 500, 1000);
|
|// Для работы таймера оставляем сценарий открытым. Получение события выполняется в фоне.
|Пока Истина Цикл
|	Приостановить(1000);
|КонецЦикла;
|";
ПутьДоФайла = ОбъединитьПути(ИОС.КаталогСценария(ЭтотОбъект), ИОС.СлучайноеИмя + ".os");
ИОС.ЗаписатьВесьТекст(ПутьДоФайла, Стр);
ИОС.ОжидатьФайл(ПутьДоФайла);

ИнформацияЗапускаПроцесса1 = ИОС.ИнформацияЗапускаПроцесса();
ИнформацияЗапускаПроцесса1.ИмяФайла = ОбъединитьПути(КаталогПрограммы(), "oscript.exe");
ИнформацияЗапускаПроцесса1.Аргументы = ПутьДоФайла;
ИнформацияЗапускаПроцесса1.ИспользоватьОболочку = Ложь;
ИнформацияЗапускаПроцесса1.ПеренаправитьВывод = Истина;

Процесс1 = ИОС.Процесс();
Процесс1.НачальнаяИнформация = ИнформацияЗапускаПроцесса1;
Процесс1.ПриПолученииДанных = ИОС.Действие(ЭтотОбъект, "Процесс1_ПриПолученииДанных");

ИОС.ОдноразовыйТаймер(ИОС.Действие(ЭтотОбъект, "ОбработкаТаймера1"), 4000);

Процесс1.Начать();
Процесс1.НачатьЧтениеВывода();

// Чтобы получить и обработать события сценарий оставим открытым.
Пока ИОС.Продолжать Цикл
	Приостановить(1000);
КонецЦикла;
